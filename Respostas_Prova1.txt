

public class NomeDaClasse {

  
    // Formato: modificador tipoDaVariavel nomeDaVariavel;
    public int variavelInteiro;
    public String variavelTexto;
    public String variavelTipo;
    public double variavelDecimal;



    // Formato: public NomeDaClasse() { }
    public NomeDaClasse() {
    }



    // Formato: public NomeDaClasse(tipo variavel, tipo variavel, ...) { }
    public NomeDaClasse(int variavelInteiro, String variavelTexto, String variavelTipo, double variavelDecimal) {
        this.variavelInteiro = variavelInteiro;
        this.variavelTexto = variavelTexto;
        this.variavelTipo = variavelTipo;
        this.variavelDecimal = variavelDecimal;
    }


   
    // Formato: public tipoRetorno NomeMetodo(parametrosOpcionais) { }
    public void NomeMetodo1() {
        System.out.println("mensagem ou acao executada");
    }

    public void NomeMetodo2() {
        System.out.println("mensagem ou acao executada");
    }
}

    }
